// ============================================================
// ejemplo_semantico.txt — Pruebas de ANALISIS SEMANTICO
// Cada bloque muestra un caso distinto: OK, ERROR o WARNING.
// ============================================================

// --- Declaraciones validas ---
int a = 10;
double pi = 3.14;
char letra = 'x';

// --- Ampliacion valida: int -> double (OK) ---
double copia = a;

// --- WARNING: truncamiento double -> int ---
int truncado = 3.99;

// --- ERROR: redeclaracion ---
int a = 20;

// --- ERROR: variable no declarada ---
int desconocido = noExiste + 1;

// --- ERROR: tipos incompatibles en declaracion ---
int textoComoInt = "hola";
int boolComoInt = true;

// --- Operadores aritmeticos validos ---
int suma = a + truncado;
double mezcla = a + pi;

// --- ERROR: operacion aritmetica invalida (bool + int) ---
int operacionInvalida = true + 1;

// --- Comparaciones y logicos validos ---
cout << (a > pi);
cout << (a > 0 && pi < 100.0);
cout << (letra == 'x');

// --- ERROR: comparacion invalida (string vs int) ---
cout << ("hola" > a);

// --- ERROR: AND con operandos no booleanos ---
cout << (a && pi);

// --- Operadores unarios y parentesis ---

// --- OK: '-' sobre char se promueve a int (C++: -'x' es int) ---
int negChar = -letra;

// --- OK: '!' sobre numerico es valido, el resultado es bool ---
cout << !a;

// --- ERROR: parentesis ahora propagan el tipo interno (antes se ignoraba) ---
int agrupado = ("hola");

// --- ERROR: '-' (negativo) requiere un operando numerico ---
int negativoInvalido = -"hola";

// --- ERROR: '!' requiere un operando booleano o numerico ---
cout << !"hola";

// --- Ambitos anidados ---

// --- OK: shadowing, no es redeclaracion porque es otro ambito ---
int sombra = 1;
if (sombra > 0) {
    int sombra = 2;
    cout << sombra;
}

// --- ERROR: soloInterna no existe afuera del bloque ---
if (sombra > 0) {
    int soloInterna = 99;
}
cout << soloInterna;

// --- Funciones ---

// --- OK: funcion con parametros, llamada y return correctos ---
int sumarDos(int x, int y) {
    return x + y;
}
int resultadoSuma = sumarDos(a, truncado);

// --- ERROR: cantidad de argumentos distinta a la declarada ---
int otroResultado = sumarDos(a);

// --- ERROR: funcion no declarada ---
int faltante = noExisteFuncion(a);

// --- ERROR: return de tipo incompatible ---
int devuelveMal(int x) {
    return "hola";
}
